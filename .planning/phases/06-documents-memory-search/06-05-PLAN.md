Per DOC-10: Create read-only document render pipeline:

1. Create `src/web/src/lib/components/docs/ReadOnlyRenderer.svelte`:
```svelte
<script lang="ts">
  import { onMount } from "svelte";
  import { unified } from "unified";
  import remarkParse from "remark-parse";
  import remarkRehype from "remark-rehype";
  import rehypeShiki from "@shikijs/rehype";
  import rehypeStringify from "rehype-stringify";
  import DOMPurify from "dompurify";

  export let markdown: string;
  let html = "";

  onMount(async () => {
    const result = await unified()
      .use(remarkParse)
      .use(remarkRehype)
      .use(rehypeShiki, { theme: "github-dark" })
      .use(rehypeStringify)
      .process(markdown);
    html = DOMPurify.sanitize(String(result));
  });
</script>

{@html html}
```

- Pipeline: remark (parse md) → rehype (to HTML) → shiki (syntax highlight) → DOMPurify (XSS sanitize)
- Used for: TUI doc viewer passthrough, read-only web view, shared docs

```typescript
describe("Read-only renderer XSS sanitization (DOC-10)", () => {
  it("strips script tags from rendered HTML", () => {
    const malicious = '<p>Hello</p><script>alert("xss")</script>';
    const clean = DOMPurify.sanitize(malicious);
    expect(clean).not.toContain("<script>");
    expect(clean).toContain("<p>Hello</p>");
  });

  it("strips onerror attributes", () => {
    const malicious = '<img src=x onerror="alert(1)">';
    const clean = DOMPurify.sanitize(malicious);
    expect(clean).not.toContain("onerror");
  });
});
```
