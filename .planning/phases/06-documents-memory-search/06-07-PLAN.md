```typescript
import { create, insert, search, count } from "@orama/orama";
import { persist, restore } from "@orama/plugin-data-persistence";

const SCHEMA = {
  title: "string",
  body: "string",
  kind: "enum",
  project: "string",
  status: "string",
  updatedAt: "number",
  entityId: "string",
} as const;

export class OramaIndex {
  private db: Awaited<ReturnType<typeof create>> | null = null;

  async hydrate(snapshot: string): Promise<void> {
    this.db = await restore("json", snapshot);
  }

  async search(term: string, options?: { facets?: boolean; where?: Record<string, unknown>; limit?: number }) {
    if (!this.db) return { hits: [], facets: {} };
    return search(this.db, {
      term,
      limit: options?.limit ?? 20,
      facets: options?.facets ? { kind: {}, project: {}, status: {} } : undefined,
      where: options?.where,
    });
  }

  async serialize(): Promise<string> {
    if (!this.db) return "";
    return persist(this.db, "json") as unknown as string;
  }
}

export const oramaIndex = new OramaIndex();
```
