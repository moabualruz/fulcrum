import { injectable as Injectable } from "@needle-di/core";

export interface ContextBundle {
  memories: unknown[];
  linkedDocs: unknown[];
  recentRuns: unknown[];
  repoState: unknown[];
  skillPrompts: unknown[];
}

export interface BundleContext {
  orgId: string;
  projectId: string;
  taskId?: string;
}

const TOTAL_TOKEN_BUDGET = 8000;
const SLICE_BUDGETS = {
  memories: 0.25,
  linkedDocs: 0.20,
  recentRuns: 0.35,
  repoState: 0.10,
  skillPrompts: 0.10,
} as const;

function estimateTokens(text: string): number {
  return Math.ceil(text.length / 4);
}

@Injectable()
export class ContextBundleService {
  async assemble(ctx: BundleContext): Promise<ContextBundle> {
    const bundle: ContextBundle = {
      memories: [],
      linkedDocs: [],
      recentRuns: [],
      repoState: [],     // Empty placeholder per D-29
      skillPrompts: [],
    };
    for (const [name, fraction] of Object.entries(SLICE_BUDGETS)) {
      if (name === "repoState") continue; // Phase 7
      const sliceBudget = Math.floor(TOTAL_TOKEN_BUDGET * fraction);
      (bundle as Record<string, unknown[]>)[name] = await this.fillSlice(name, ctx, sliceBudget);
    }
    return bundle;
  }

  private async fillSlice(name: string, ctx: BundleContext, budget: number): Promise<unknown[]> {
    const items = await this.retrieveSliceItems(name, ctx);
    const result: unknown[] = [];
    let used = 0;
    for (const item of items) {
      const cost = estimateTokens(JSON.stringify(item));
      if (used + cost > budget) break;
      result.push(item);
      used += cost;
    }
    return result;
  }

  private async retrieveSliceItems(name: string, ctx: BundleContext): Promise<unknown[]> {
    // Each slice calls its retriever — inject via constructor or lazy import
    // memories: MemoryRepository.searchProjectAndGlobal()
    // linkedDocs: Document context_summaries for task's linked docs
    // recentRuns: recent AgentRun results for project
    // skillPrompts: active skill prompt text
    return [];
  }
}
