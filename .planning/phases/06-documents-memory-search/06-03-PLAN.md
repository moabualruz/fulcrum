import { injectable as Injectable, inject as Inject } from "@needle-di/core";
import type { MemoryRepository } from "@/db/repositories/MemoryRepository";
import type { DocumentRepository } from "@/db/repositories/DocumentRepository";
import type { AgentRunRepository } from "@/db/repositories/AgentRunRepository";

export interface ContextBundle {
  memories: unknown[];
  linkedDocs: unknown[];
  recentRuns: unknown[];
  repoState: unknown[];
  skillPrompts: unknown[];
}

export interface BundleContext {
  orgId: string;
  projectId: string;
  taskId?: string;
}

const TOTAL_TOKEN_BUDGET = 8000;
const SLICE_BUDGETS = {
  memories: 0.25,
  linkedDocs: 0.20,
  recentRuns: 0.35,
  repoState: 0.10,
  skillPrompts: 0.10,
} as const;

function estimateTokens(text: string): number {
  return Math.ceil(text.length / 4);
}

@Injectable()
export class ContextBundleService {
  constructor(
    @Inject("MemoryRepository") private memoryRepo: MemoryRepository,
    @Inject("DocumentRepository") private documentRepo: DocumentRepository,
    @Inject("AgentRunRepository") private agentRunRepo: AgentRunRepository,
  ) {}

  async assemble(ctx: BundleContext): Promise<ContextBundle> {
    const bundle: ContextBundle = {
      memories: [],
      linkedDocs: [],
      recentRuns: [],
      repoState: [],     // Empty placeholder per D-29
      skillPrompts: [],
    };
    // memories slice
    bundle.memories = await this.fillSlice(
      await this.memoryRepo.searchProjectAndGlobal(ctx.orgId, ctx.projectId),
      Math.floor(TOTAL_TOKEN_BUDGET * SLICE_BUDGETS.memories),
    );
    // linkedDocs slice
    bundle.linkedDocs = await this.fillSlice(
      await this.documentRepo.getContextSummariesForProject(ctx.projectId),
      Math.floor(TOTAL_TOKEN_BUDGET * SLICE_BUDGETS.linkedDocs),
    );
    // recentRuns slice
    bundle.recentRuns = await this.fillSlice(
      await this.agentRunRepo.getRecentForProject(ctx.projectId),
      Math.floor(TOTAL_TOKEN_BUDGET * SLICE_BUDGETS.recentRuns),
    );
    // repoState: empty per D-29 (Phase 7)
    // skillPrompts: load active skill prompts for org
    bundle.skillPrompts = await this.fillSlice(
      await this.loadSkillPrompts(ctx.orgId),
      Math.floor(TOTAL_TOKEN_BUDGET * SLICE_BUDGETS.skillPrompts),
    );
    return bundle;
  }

  private async fillSlice(items: unknown[], budget: number): Promise<unknown[]> {
    const result: unknown[] = [];
    let used = 0;
    for (const item of items) {
      const cost = estimateTokens(JSON.stringify(item));
      if (used + cost > budget) break;
      result.push(item);
      used += cost;
    }
    return result;
  }
