<objective>
Unify three event mechanisms into a single EventDispatcher.

Purpose: Three incompatible mechanisms exist: (1) MikroORM Event entity with UUID PKs, (2) appendEvent() raw SQL with ULID PKs, (3) EventBus singleton with zero publishers. This creates PK format inconsistency and means WebSocket subscriptions never fire.

Output: Single EventDispatcher that persists (UUID) + publishes (EventBus). appendEvent and RoutingEventBus removed.
</objective>

`src/events/EventDispatcher.ts`:
```typescript
import { EntityManager } from "@mikro-orm/core";
import { Event } from "../db/entities/core/Event.ts";
import type { DomainEvent } from "./types.ts";

// Import the existing process-singleton EventBus for WebSocket publishing
import { eventBus } from "../subscriptions/event-bus.ts";

export class EventDispatcher {
  constructor(private readonly em: EntityManager) {}

  /**
   * Persist event to DB (UUID PK via Event entity) then publish to EventBus
   * for WebSocket subscribers. Single atomic operation.
   */
  async dispatch(event: DomainEvent): Promise<string> {
    const entity = this.em.create(Event, {
      kind: event.kind,
      orgId: event.orgId,
      entityId: event.entityId,
      entityType: event.entityType,
      payload: event.payload,
      userId: event.userId,
      projectId: event.projectId,
    });
    await this.em.persistAndFlush(entity);

    // Publish to EventBus for WebSocket subscriptions
    eventBus.emit(event.kind, {
      id: entity.id,  // UUID from entity
      ...event,
    });

    return entity.id;
  }

  /** Dispatch multiple events in a single flush */
  async dispatchBatch(events: DomainEvent[]): Promise<string[]> {
    const entities = events.map(e => this.em.create(Event, {
      kind: e.kind,
      orgId: e.orgId,
      entityId: e.entityId,
      entityType: e.entityType,
      payload: e.payload,
      userId: e.userId,
      projectId: e.projectId,
    }));
    await this.em.persistAndFlush(entities);

    for (const [i, entity] of entities.entries()) {
      eventBus.emit(events[i].kind, { id: entity.id, ...events[i] });
    }

    return entities.map(e => e.id);
  }
}
```
