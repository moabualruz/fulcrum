<objective>
Replace per-request PGlite connection opening with a single startup connection pool.

Purpose: 77 page.server.ts files call openProductDb() which opens PGlite + runs migrations on EVERY page load. This causes ~500ms overhead per request and potential data corruption from concurrent migrations.

Output: Single PGlite instance initialized at startup, shared via getProductDb().
</objective>

Rewrite `src/web/src/lib/server/db.ts` to use a singleton pattern:

```typescript
import { join } from "node:path";
import { openPglite } from "../../../../product-kernel/db/pglite.ts";
import { runMigrations } from "../../../../product-kernel/db/migrate.ts";
import { productDbDir } from "../../../../product-kernel/paths.ts";
import type { ProductDb } from "../../../../product-kernel/db/types.ts";

let _db: ProductDb | null = null;
let _initPromise: Promise<ProductDb> | null = null;

/**
 * Initialize PGlite once at startup. Runs migrations + ensures default org.
 * Call from hooks.server.ts or server init, NOT per-request.
 */
export async function initProductDb(): Promise<ProductDb> {
  if (_db) return _db;
  if (_initPromise) return _initPromise;
  _initPromise = (async () => {
    const db = await openPglite(join(productDbDir(), "main"));
    await runMigrations(db);
    await ensureDefaultOrg(db);
    _db = db;
    return db;
  })();
  return _initPromise;
}

/**
 * Get the shared PGlite instance. Throws if not initialized.
 * For use in page.server.ts load functions.
 */
export function getProductDb(): ProductDb {
  if (!_db) throw new Error("ProductDb not initialized — call initProductDb() at startup");
  return _db;
}

/**
 * @deprecated Use getProductDb() instead. This opens a NEW connection per call.
 * Kept temporarily for migration — will be removed after all callers updated.
 */
export async function openProductDb(): Promise<ProductDb> {
  // During migration: just return the singleton if available
  if (_db) return _db;
  return initProductDb();
}
```

Keep the existing `getDefaultOrgId` and `ensureDefaultOrg` functions unchanged.

This approach:
1. Keeps `openProductDb` working (returns singleton) so 77 callers don't break immediately
2. Adds `getProductDb()` sync getter for new code
3. Adds `initProductDb()` for startup initialization
