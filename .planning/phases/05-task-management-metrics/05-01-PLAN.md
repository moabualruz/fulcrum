1. `task_comments` — id(uuid PK), org_id(FK orgs cascade), task_id(uuid NOT NULL), author_id(uuid NOT NULL), body(jsonb default '{}'), parent_comment_id(uuid NULL self-FK cascade), resolved(bool default false), resolved_by(uuid NULL), resolved_at(timestamptz NULL), created_at, updated_at. Indexes: (task_id), (org_id, task_id).

2. `comment_reactions` — id(uuid PK), comment_id(FK task_comments cascade), user_id(uuid NOT NULL), emoji(varchar(8) NOT NULL), created_at. UNIQUE(comment_id, user_id, emoji).

3. `task_watchers` — id(uuid PK), org_id(FK orgs cascade), task_id(uuid NOT NULL), user_id(uuid NOT NULL), source(varchar check in ('manual','mention','assign','create') default 'manual'), created_at. UNIQUE(task_id, user_id).

4. `task_relationships` — id(uuid PK), org_id(FK orgs cascade), source_task_id(uuid NOT NULL), target_task_id(uuid NOT NULL), type(varchar check in ('blocks','relates_to','duplicate_of') NOT NULL), created_by(uuid NOT NULL), created_at. UNIQUE(source_task_id, target_task_id, type). Indexes: (source_task_id), (target_task_id).

5. `project_automations` — id(uuid PK), org_id(FK orgs cascade), project_id(uuid NOT NULL), name(varchar NOT NULL), trigger_type(varchar NOT NULL), trigger_config(jsonb default '{}'), condition(jsonb NULL), action_type(varchar NOT NULL), action_config(jsonb default '{}'), enabled(bool default true), execution_count(int default 0), created_at, updated_at. Index: (project_id, enabled).

6. `field_dependency_rules` — id(uuid PK), org_id(FK orgs cascade), project_id(uuid NOT NULL), source_field_id(uuid NOT NULL), source_value(varchar NOT NULL), target_field_id(uuid NOT NULL), action(varchar check in ('show','hide','require') NOT NULL), created_at. Index: (project_id).

7. `yjs_snapshots` (HIGH-05 fix — in Wave 1) — id(uuid PK), doc_name(varchar NOT NULL UNIQUE), state(bytea NOT NULL), updated_at(timestamptz default now()).
