**Import tinykeys** and register all D-67 shortcuts in `$effect`:
```typescript
import tinykeys from "tinykeys";

$effect(() => {
  const unsubscribe = tinykeys(window, {
    "$mod+KeyK": (e) => { e.preventDefault(); commandPaletteOpen = true; },
    "c": () => { if (!isEditing()) createTask(); },
    "j": () => { if (!isEditing()) navigateTask("next"); },
    "k": () => { if (!isEditing()) navigateTask("prev"); },
    "Enter": () => { if (!isEditing()) openSelectedTask(); },
    "Escape": () => { closePanel(); },
    "s": () => { if (!isEditing()) openStatusPicker(); },
    "a": () => { if (!isEditing()) openAssigneePicker(); },
    "p": () => { if (!isEditing()) openPriorityPicker(); },
    "l": () => { if (!isEditing()) openLabelPicker(); },
    "m": () => { if (!isEditing()) openSprintPicker(); },
    "x": () => { if (!isEditing()) toggleSelection(); },
    "Shift+KeyC": () => { if (!isEditing()) addToCurrentSprint(); },
    "?": () => { shortcutOverlayOpen = true; },
  });
  return unsubscribe;
});
```

**Guard function** `isEditing()`: return true if document.activeElement is input/textarea/contenteditable — prevents shortcuts firing during text entry.
