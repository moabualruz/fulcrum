Create `src/server/trpc/routers/comments.ts` with tRPC router:

```typescript
export const commentsRouter = t.router({
  list: permissionedProcedure({ resource: "comments", action: "list" })
    .input(z.object({ taskId: z.string().uuid() }))
    .query(/* delegate to CommentService.listComments */),

  create: permissionedProcedure({ resource: "comments", action: "create" })
    .input(z.object({ taskId: z.string().uuid(), body: z.record(z.unknown()), parentCommentId: z.string().uuid().optional() }))
    .mutation(/* delegate to CommentService.createComment */),

  delete: permissionedProcedure({ resource: "comments", action: "delete" })
    .input(z.object({ commentId: z.string().uuid() }))
    .mutation(/* delegate to CommentService.deleteComment */),

  resolve: permissionedProcedure({ resource: "comments", action: "update" })
    .input(z.object({ commentId: z.string().uuid() }))
    .mutation(/* delegate to CommentService.resolveComment */),

  unresolve: permissionedProcedure({ resource: "comments", action: "update" })
    .input(z.object({ commentId: z.string().uuid() }))
    .mutation(/* delegate to CommentService.unresolveComment */),

  addReaction: permissionedProcedure({ resource: "comments", action: "update" })
    .input(z.object({ commentId: z.string().uuid(), emoji: z.string().max(8) }))
    .mutation(/* delegate to CommentService.addReaction */),

  removeReaction: permissionedProcedure({ resource: "comments", action: "update" })
    .input(z.object({ commentId: z.string().uuid(), emoji: z.string().max(8) }))
    .mutation(/* delegate to CommentService.removeReaction */),

  // Watcher procedures
  watchers: permissionedProcedure({ resource: "comments", action: "list" })
    .input(z.object({ taskId: z.string().uuid() }))
    .query(/* delegate to CommentService.listWatchers */),

  subscribe: permissionedProcedure({ resource: "comments", action: "create" })
    .input(z.object({ taskId: z.string().uuid() }))
    .mutation(/* delegate to CommentService.subscribe with source 'manual' */),

  unsubscribe: permissionedProcedure({ resource: "comments", action: "delete" })
    .input(z.object({ taskId: z.string().uuid() }))
    .mutation(/* delegate to CommentService.unsubscribe */),
});
```

Do NOT modify `src/trpc/router.ts` here — Plan 06 owns that file exclusively (HIGH-06 fix).
