- tableName: "task_comments"
- Indexes: task_id, (org_id + task_id)
- Props: id, org(ManyToOne Org), taskId(uuid), authorId(uuid), body(json, TipTap JSON), parentCommentId(uuid nullable), resolved(bool default false), resolvedBy(uuid nullable), resolvedAt(datetime nullable), createdAt, updatedAt
- OptionalProps: createdAt, updatedAt, resolved, resolvedAt, resolvedBy, parentCommentId, body

**TaskWatcher.ts** (D-07, D-08):
- tableName: "task_watchers"
- Unique: [taskId, userId]
- Props: id, org(ManyToOne), taskId(uuid), userId(uuid), source(string — 'manual'|'mention'|'assign'|'create'), createdAt
- OptionalProps: createdAt, source

**CommentReaction.ts** (D-04):
- tableName: "comment_reactions"
- Unique: [commentId, userId, emoji]
- Props: id, commentId(uuid), userId(uuid), emoji(string), createdAt
- OptionalProps: createdAt

**TaskRelationship.ts** (D-19):
- tableName: "task_relationships"
- Unique: [sourceTaskId, targetTaskId, type]
- Indexes: source_task_id, target_task_id
- Props: id, org(ManyToOne), sourceTaskId(uuid), targetTaskId(uuid), type(string — 'blocks'|'relates_to'|'duplicate_of'), createdBy(uuid), createdAt
- OptionalProps: createdAt

**ProjectAutomation.ts** (D-89, D-90):
- tableName: "project_automations"
- Index: (project_id, enabled)
- Props: id, org(ManyToOne), projectId(uuid), name(string), triggerType(string), triggerConfig(json), condition(json nullable), actionType(string), actionConfig(json), enabled(bool default true), executionCount(integer default 0), createdAt, updatedAt
- OptionalProps: createdAt, updatedAt, enabled, executionCount, triggerConfig, actionConfig

**FieldDependencyRule.ts** (D-109):
- tableName: "field_dependency_rules"
- Index: project_id
- Props: id, org(ManyToOne), projectId(uuid), sourceFieldId(uuid), sourceValue(string), targetFieldId(uuid), action(string — 'show'|'hide'|'require'), createdAt
- OptionalProps: createdAt

- Props: id, docName(string, unique), state(Buffer/Uint8Array — type "blob"), updatedAt
